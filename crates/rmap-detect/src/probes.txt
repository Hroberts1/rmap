# rmap service probe database.
#
# Written for rmap from protocol documentation and observed banners. The
# format is described in crates/rmap-detect/src/db.rs and README.md.
#
# Probe order matters: on a port, probes that list it in `ports` run first
# (in file order), then `generic` probes. Within a probe, the first matching
# rule wins, so put specific rules before general ones. Every rule needs a
# test banner in crates/rmap-detect/src/rule_tests.rs.

# ----------------------------------------------------------------- banner ---
# Many services speak first. Send nothing and read.
probe banner
  ports    21,22,23,25,110,143,587,2121,2222,3306,5900-5903
  wait     3s
  generic

  product  OpenSSH
  version  $2
  info     protocol $1
  os       Windows
match ssh ^SSH-([\d.]+)-OpenSSH_([\w.]+) (Ubuntu-\S+)
  product  OpenSSH
  version  $2
  info     protocol $1; $3
  os       Ubuntu Linux
match ssh ^SSH-([\d.]+)-OpenSSH_([\w.]+) (Debian-\S+)
  product  OpenSSH
  version  $2
  info     protocol $1; $3
  os       Debian Linux
match ssh ^SSH-([\d.]+)-OpenSSH_([\w.]+) (Raspbian-\S+)
  product  OpenSSH
  version  $2
  info     protocol $1; $3
  os       Raspbian Linux
match ssh ^SSH-([\d.]+)-OpenSSH_([\w.]+) (FreeBSD-\S+)
  product  OpenSSH
  version  $2
  info     protocol $1; $3
  os       FreeBSD
match ssh ^SSH-([\d.]+)-OpenSSH_([\w.]+)(?: ([^\r\n]+))?

  version  $2
  info     protocol $1
match ssh ^SSH-([\d.]+)-([^\r\n]+)
  info     protocol $1; $2
  confidence 7

# --- MySQL and MariaDB greeting: 3-byte length, sequence 0, protocol 10,
# then a NUL-terminated version string. MariaDB prefixes "5.5.5-".
match mysql ^[\s\S]{3}\x00\x0a(?:5\.5\.5-)?([\d.]+)-([^\x00]*MariaDB[^\x00]*(?:ubuntu|ubu\d)[^\x00]*)\x00
  product  MariaDB
  version  $1
  info     $2
  os       Ubuntu Linux
match mysql ^[\s\S]{3}\x00\x0a(?:5\.5\.5-)?([\d.]+)-([^\x00]*MariaDB[^\x00]*deb[^\x00]*)\x00
  product  MariaDB
  version  $1
  info     $2
  os       Debian Linux
match mysql ^[\s\S]{3}\x00\x0a(?:5\.5\.5-)?([\d.]+)-([^\x00]*MariaDB[^\x00]*)\x00
  product  MariaDB
  version  $1
  info     $2
match mysql ^[\s\S]{3}\x00\x0a([\d.]+)(-[^\x00]*ubuntu[^\x00]*)\x00
  product  MySQL
  version  $1
  info     $2
  os       Ubuntu Linux
match mysql ^[\s\S]{3}\x00\x0a([4-9]\.[\d.]+)([^\x00]*)\x00
  product  MySQL
  version  $1
  info     $2
match mysql ^[\s\S]{3}\x00\xff[\s\S]{2}(?:#\w{5})?Host '[^']*' is not allowed to connect to this (MySQL|MariaDB) server
  product  $1
  info     host not allowed to connect
match mysql ^[\s\S]{3}\x00\xff\x10\x04Too many connections
  info     too many connections
  confidence 7
